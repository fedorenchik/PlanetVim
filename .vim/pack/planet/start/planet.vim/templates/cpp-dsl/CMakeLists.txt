cmake_minimum_required(VERSION 3.16)
project(planet_dsl LANGUAGES CXX)
add_executable(planet_dsl main.cpp)
target_compile_features(planet_dsl PRIVATE cxx_std_17)
enable_testing()
add_test(NAME sum COMMAND planet_dsl "1 + 2")
set_tests_properties(sum PROPERTIES PASS_REGULAR_EXPRESSION "3")
add_test(NAME invalid COMMAND planet_dsl "1 + bad")
set_tests_properties(invalid PROPERTIES WILL_FAIL TRUE)
